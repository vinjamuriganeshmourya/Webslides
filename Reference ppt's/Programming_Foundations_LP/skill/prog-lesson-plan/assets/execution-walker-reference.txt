EXECUTION WALKER — worked reference
===================================

Lifted verbatim from session 11 ("Follow the Dot"), the plan that works. Adapt the RUNGS /
POS tables and the flowchart geometry to your topic; leave the state machine alone.

Read references/animated-explainer.md first — it has the three tests this must pass before
you build it, and the reasons behind the rules below.

Four things not to change:
  1. .dead persists after the run ends. It is the finding, not a transition state.
  2. plan() computes the steps from the live input. A hard-coded step list makes the
     second walk a lie.
  3. One moving element (the dot). Two desynchronise and the room watches the wrong one.
  4. The @media (prefers-reduced-motion) block, and the fact that print shows the END state
     with dead paths still grey.

The output box starts empty on purpose: `.wout:empty::before` carries the placeholder, so
the room predicts before anything prints.

3. STEP ENGINE  (inside the template's main IIFE)
------------------------------------------------------------------------------

/* ================= execution walker =================
     Steps are computed from the slider, so the room can change the input and
     walk it again. Only executed lines highlight; at the end everything the
     interpreter never touched greys out and stays grey. */
  (function(){
    var W=document.getElementById('walker');
    if(!W) return;
    var RUNGS=[[91,'A'],[81,'B'],[71,'C'],[61,'D']];
    var POS={a:[210,34],d0:[210,105],p0:[450,105],d1:[210,205],p1:[450,205],
             d2:[210,305],p2:[450,305],d3:[210,405],p3:[450,405],e:[210,505],pf:[450,505]};
    var dot=W.querySelector('.wdot'),
        out=W.querySelector('.wout'),
        slider=W.querySelector('input[type=range]'),
        label=W.querySelector('.wslide label'),
        playBtn=W.querySelector('[data-w=play]'),
        stepBtn=W.querySelector('[data-w=step]'),
        resetBtn=W.querySelector('[data-w=reset]');
    var steps=[], at=0, timer=null;

    function plan(m){
      // one step per thing Python actually does, in order
      var s=[{node:'a', line:1, say:'marks = '+m}];
      for(var i=0;i<RUNGS.length;i++){
        var hit = m >= RUNGS[i][0];
        s.push({node:'d'+i, line:3+i*2, test:hit, edge:'d'+i+'in'});
        if(hit){
          s.push({node:'p'+i, line:4+i*2, edge:'t'+i, print:RUNGS[i][1]});
          return s;
        }
      }
      s.push({node:'e', line:11, edge:'d4in'});
      s.push({node:'pf', line:12, edge:'tf', print:'F'});
      return s;
    }

    function clear(){
      W.querySelectorAll('.wline').forEach(function(l){ l.className='wline'; });
      W.querySelectorAll('[data-node],[data-edge]').forEach(function(n){
        n.classList.remove('visited','dead','no'); });
      dot.classList.remove('off');
      dot.style.transform='translate(210px,34px)';
      out.textContent='';
    }

    function render(){
      var s=steps[at-1];
      if(!s) return;
      W.querySelectorAll('.wline.now').forEach(function(l){ l.classList.remove('now'); });
      var line=W.querySelector('#wl'+s.line);
      if(line){ line.classList.add('ran','now'); }
      var node=W.querySelector('[data-node="'+s.node+'"]');
      if(node){ node.classList.add('visited'); if(s.test===false) node.classList.add('no'); }
      if(s.edge){
        var e=W.querySelector('[data-edge="'+s.edge+'"]');
        if(e) e.classList.add('visited');
      }
      var p=POS[s.node];
      dot.style.transform='translate('+p[0]+'px,'+p[1]+'px)';
      if(s.print) out.textContent=s.print;
      if(at===steps.length) finish();
    }

    function finish(){
      // grey out every line and node the dot never reached, and leave it grey
      W.querySelectorAll('.wline').forEach(function(l){
        if(!l.classList.contains('ran') && l.textContent.trim()) l.classList.add('dead');
      });
      W.querySelectorAll('[data-node],[data-edge]').forEach(function(n){
        if(!n.classList.contains('visited')) n.classList.add('dead');
      });
      W.querySelectorAll('.wline.now').forEach(function(l){ l.classList.remove('now'); });
    }

    function step(){
      if(at>=steps.length) return false;
      at++; render(); return at<steps.length;
    }
    function stop(){ if(timer){ clearInterval(timer); timer=null; }
      playBtn.textContent='Play'; }
    function play(){
      if(timer){ stop(); return; }
      if(at>=steps.length) reset();
      playBtn.textContent='Pause';
      timer=setInterval(function(){ if(!step()) stop(); }, 780);
    }
    function reset(){ stop(); at=0; steps=plan(+slider.value); clear(); }

    slider.addEventListener('input', function(){
      label.textContent='marks = '+slider.value;
      var l=W.querySelector('#wl1');
      if(l) l.innerHTML='marks <b>=</b> <i>'+slider.value+'</i>';
      var a=W.querySelector('#wassign');          // the flowchart's first node tracks it too
      if(a) a.textContent='marks = '+slider.value;
      reset();
    });
    playBtn.addEventListener('click', play);
    stepBtn.addEventListener('click', function(){ stop(); if(at>=steps.length) reset(); step(); });
    resetBtn.addEventListener('click', reset);
    reset();

    // r replays the walker when its card is live, same key as the hook
    W.replayWalker = reset;
  })();

4. ADAPTING IT
------------------------------------------------------------------------------

For a while loop, the shape changes but nothing else does:

  - RUNGS becomes the loop's passes, computed from the slider, not listed.
  - plan() emits, per pass: condition-test node, then each body line, then the update —
    and re-emits the condition test. That re-emission IS the lesson: the room sees the
    condition asked again between passes.
  - The dot returns UP the back edge. That edge must be a real <path data-edge="back">
    so it can light up; an arrow drawn in the background image cannot.
  - The final condition test is the one that goes .no and ends the walk, and the body
    lines below it never go .dead — they ran. What goes dead is print("End") on an
    infinite-loop variant, which is the version worth walking second.

For a for loop, POS tracks the item cursor over the sequence cells rather than a route
down a page; the greying-out applies to cells not yet consumed.

Whatever the topic: one dot, steps computed from a live input, dead state persists.
